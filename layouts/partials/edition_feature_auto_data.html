"论文列表" -}} {{- if $inList }}{{ $sections = add $sections 1 }}{{ end -}} {{- else if and $inList (findRE `^###[ \t]+` . 1) -}} {{- $links := findRE `\]\(/[^)]+\)` . -}} {{- if eq (len $links) 1 -}} {{- $target := strings.TrimPrefix "](" (strings.TrimSuffix ")" (index $links 0)) -}} {{- $route := printf "/%s" (strings.TrimPrefix "/" (strings.TrimPrefix site.Home.RelPermalink $target)) -}} {{- if eq (strings.TrimSuffix "/" $route) (strings.TrimSuffix "/" $paperPath) }}{{ $members = add $members 1 }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- $imageSource := $record.sourceImageUrl | default "" -}} {{- $imageAlt := $record.imageAlt | default "" -}} {{- $imageMarkdown := $record.imageMarkdown | default "" -}} {{- $validImage := and (eq (len (findRE `^!\[(\\.|[^\]\\\n])*\]\([^ \t\r\n)]+\)$` $imageMarkdown 1)) 1) (strings.HasSuffix $imageMarkdown (printf "](%s)" $imageSource)) (eq (crypto.SHA256 $imageMarkdown) ($record.imageMarkdownSha256 | default "")) -}} {{- $valid = and $valid (eq $sections 1) (eq $members 1) $validImage (ne $imageAlt "") (strings.Contains $paper.RawContent $imageMarkdown) (ne ($record.title | default "") "") -}} {{- $imageUrl := "" -}} {{- $officialPrefix := printf "https://arxiv.org/html/%s/" $version -}} {{- $officialRelative := strings.TrimPrefix $officialPrefix $imageSource -}} {{- $officialSource := and (strings.HasPrefix $imageSource $officialPrefix) (eq (len (findRE `(?i)^[A-Za-z0-9_./~-]+\.(png|jpg|jpeg|webp|svg)$` $officialRelative 1)) 1) (not (findRE `(^|/)\.{1,2}(/|$)` $officialRelative 1)) -}} {{- if eq ($record.imageKind | default "") "official" -}} {{- /* This is the same official URL already used in the reviewed article; (eq $mappedPath $imagePath) -}} {{- end -}} {{- $valid = and $sourceMatches (eq (crypto.SHA256 (readFile $imageFile)) ($record.imageSha256 | default "")) (gt (int ($record.width | default 0)) 0) (gt (int ($record.height | default 0)) 0) -}} {{- end -}} {{- else -}} {{- $valid = false -}} {{- end -}} {{- if $valid }}{{ $result = merge $record (dict "paperUrl" $paper.RelPermalink "imageUrl" $imageUrl "automatic" true) }}{{ end -}} {{- end -}} {{- end -}} {{- return $result -}}
